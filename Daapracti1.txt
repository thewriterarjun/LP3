Problem Statement :Write a program non-recursive and recursive program to
calculate Fibonacci numbers and
analyze their time and space complexity.


1.Recursive
#include<iostream>
using namespace std;
int fibonacci(int n)
{
if(n<=1)
return n;
return fibonacci(n-1)+fibonacci(n-2);
}
int main()
{
int n;
cout<<"Enter the number of terms : ";
cin>>n;
for(int i=0;i<n;i++)
{
cout<<fibonacci(i)<<" ";
}
return 0;
}

2.Non Recursive
#include<iostream>
using namespace std;
int main()
{
int n;
cout<<"Enter the number of terms : ";
cin>>n;
int a=0,b=1;
cout<<"\nFibonacci Series: ";
for(int i=0;i<n;i++)
{
cout<<a<<" ";
int c=a+b;
a=b;
b=c;
}
return 0;
}
