Problem Statement : Write a program to implement Huffman Encoding using a
greedy strategy.

Code:

import java.util.*;

class Node {
    char data;
    int freq;
    Node left, right;

    Node(char d, int f) {
        data = d;
        freq = f;
    }
}

public class Main {

    static void generateCodes(Node root, String code) {
        if (root == null) return;

        if (root.left == null && root.right == null) {
            System.out.println(root.data + " : " + code);
            return;
        }

        generateCodes(root.left, code + "0");
        generateCodes(root.right, code + "1");
    }

    static void huffmanEncoding(char[] data, int[] freq) {
        PriorityQueue<Node> pq =
            new PriorityQueue<>((a, b) -> a.freq - b.freq);

        for (int i = 0; i < data.length; i++)
            pq.add(new Node(data[i], freq[i]));

        while (pq.size() > 1) {
            Node l = pq.poll(), r = pq.poll();
            Node n = new Node('\0', l.freq + r.freq);
            n.left = l;
            n.right = r;
            pq.add(n);
        }

        System.out.println("Huffman Codes:");
        generateCodes(pq.poll(), "");
    }

    public static void main(String[] args) {
        char[] data = {'A', 'B', 'C', 'D', 'E', 'F'};
        int[] freq = {5, 9, 12, 13, 16, 45};

        huffmanEncoding(data, freq);
    }
}
